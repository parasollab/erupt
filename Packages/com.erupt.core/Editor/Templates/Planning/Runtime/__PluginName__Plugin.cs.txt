        private RobotSessions<Session> sessions;

        public override string Id => "{{PluginId}}";

        /// <summary>Robots without a session (no ROS bus) are left to another planner.</summary>
        public override bool Serves(RobotInstance robot) => sessions?.For(robot) != null;

        protected override void OnRegister(IEruptContext context)
        {
            // One session per robot, now and for every robot that joins later. Return null for
            // a robot this planner cannot serve.
            sessions = new RobotSessions<Session>(context.Robots,
                robot => robot.Ros != null ? new Session(robot) : null);
            base.OnRegister(context);       // binds the verbs and adds the settings tab
        }

        protected override void OnUnregister(IEruptContext context)
        {
            sessions?.Dispose();            // disposes every robot's session
            sessions = null;
            base.OnUnregister(context);     // destroys this plugin's trajectory selectables
        }

        /// <summary>The user chose a robot's end effector and pressed set-goal. Return a refusal with a reason when you cannot take one.</summary>
        public override InteractionRefusal SetGoal(RobotInstance robot, ISelectable endEffector)
        {
